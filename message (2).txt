local function isInLobby()
    local C = LocalPlayer.Character
    if not C then return false end
    if LocalPlayer:GetAttribute("LobbyTraining") then return false end
    local il = false
    if C:GetAttribute("Stunned") then il = true end
    local aliveFolder = workspace:FindFirstChild("Alive")
    local lobbyParryAttr = LocalPlayer:GetAttribute("LobbyParry")
    if aliveFolder and C.Parent ~= aliveFolder and not (DebugFlags.LobbyParry or lobbyParryAttr) then
        il = true
    end
    local qi
    pcall(function()
        if LocalPlayer.Upgrades then qi = LocalPlayer.Upgrades["Qi-Charge"] end
    end)
    if C:GetAttribute("DoNotParry") or (C:GetAttribute("ChargingAdrenaline") and qi and qi.Value < 2) then
        il = true
    end
    if lobbyParryAttr and LocalPlayer:GetAttribute("InLobbyParryCooldown") then il = true end
    return il
end

local function is_player_alive_and_in_alive()
    local char = LocalPlayer.Character
    if not char then return false end
    if LocalPlayer:GetAttribute("LobbyTraining") then return true end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local aliveFolder = workspace:FindFirstChild("Alive")
    if not aliveFolder then return false end
    if not aliveFolder:FindFirstChild(LocalPlayer.Name) then return false end
    return true
end

local function hasBallAnywhere()
    local function scan(c)
        if not c then return false end
        for _, b in ipairs(c:GetChildren()) do
            if b:GetAttribute("realBall") then return true end
        end
        return false
    end
    if scan(workspace:FindFirstChild("Balls")) then return true end
    if scan(workspace:FindFirstChild("TrainingBalls")) then return true end
    return false
end

local AnimationFixService = { Cache = {} }
local function GetParryCD() return 0 end
local function StopAnimation(animationtrack)
    local StopFadeTime = animationtrack:GetAttribute("StopFadeTime")
    animationtrack:Stop(StopFadeTime)
end

local function PlayGrabAnimation(animationtrack)
    local timesParried = ReplionDATA and ReplionDATA:Get("timesParried") or 0
    local IsVaild = (timesParried and timesParried >= 0 and timesParried <= 4) and true or false
    local PlayFadeTime = IsVaild and 0.05 or animationtrack:GetAttribute("PlayFadeTime")
    local PlayWeight = IsVaild and 1 or animationtrack:GetAttribute("PlayWeight")
    local PlaySpeed = IsVaild and (timesParried / 5 + 1) or animationtrack:GetAttribute("PlaySpeed")
    animationtrack:Play(PlayFadeTime, PlayWeight, PlaySpeed)
end

local function GetParryAnimation(swordName)
    local character = LocalPlayer.Character
    if not character then return nil end
    local cacheKey = swordName or ""
    if AnimationFixService.Cache[cacheKey] then return AnimationFixService.Cache[cacheKey] end
    local animationType = "Default"
    local swordType = "Single"
    if swordName and swordName ~= "" and Achaotic_SwordInstances then
        local ok, swordData = pcall(function() return Achaotic_SwordInstances:GetSword(swordName) end)
        if ok and swordData and type(swordData) == "table" then
            animationType = swordData.AnimationType or animationType
            swordType = swordData.SwordType or swordType
        end
    end
    local animation
    if Achaotic_SwordAPI and type(Achaotic_SwordAPI.GetAnimations) == "function" then
        local ok, animations = pcall(function()
            return Achaotic_SwordAPI:GetAnimations(character, {"Parry", "GrabParry"}, animationType, swordType)
        end)
        if ok and type(animations) == "table" then
            for _, anim in pairs(animations) do
                if typeof(anim) == "Instance" then animation = anim break end
            end
        end
    end
    if not animation then
        local defaultCollection = SwordAPI.Collection.Default
        animation = defaultCollection and (defaultCollection:FindFirstChild("GrabParry") or defaultCollection:FindFirstChild("Grab"))
    end
    AnimationFixService.Cache[cacheKey] = animation
    return animation
end

local ParryLocked = false
local ParryCooldownActive = false
local ParryProcessing = false
local ParryM1Stop = false

local function PlayParry_Animation()
    if not config.animfix then return end
    if isInLobby() then return end
    local humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    local animator = humanoid:FindFirstChildOfClass("Animator") or humanoid:FindFirstChild("Animator")
    if not animator then return end
    local currentSword = nil
    if skinChangerEnabled and skinChangerModel ~= "" then
        currentSword = skinChangerModel
    else
        local character = LocalPlayer.Character
        if not character then return end
        currentSword = character:GetAttribute("CurrentlyEquippedSword")
    end
    local animation = GetParryAnimation(currentSword)
    if not animation then
        animation = SwordAPI.Collection.Default:FindFirstChild("GrabParry")
        if not animation then return end
    end
    if AnimationController then
        for _, track in pairs(AnimationController:GetPlayingAnimationTracks(animator)) do
            if track:GetAttribute("SuccessParry") then pcall(StopAnimation, track) end
        end
        local ok, GrabParry = pcall(function() return AnimationController:LoadAnimation(animator, animation, true) end)
        if ok and GrabParry then PlayGrabAnimation(GrabParry) end
    else
        local track = animator:LoadAnimation(animation)
        track.Priority = Enum.AnimationPriority.Action4
        track:Play(0, 1, 1)
    end
end

local function SpamParry_Animation()
    if not config.animfix then return end
    if isInLobby() then return end
    if ParryProcessing or ParryCooldownActive or ParryM1Stop then return end
    ParryProcessing = true
    ParryCooldownActive = true
    PlayParry_Animation()
    local ParryCD = GetParryCD()
    task.delay(ParryCD, function()
        ParryCooldownActive = false
        task.wait(math.max(0.1, 0.3 - ParryCD))
        if not ParryLocked then ParryProcessing = false end
    end)
end

local function SetupSwordParryHook()
    if SwordChanger.hookedParryAll then return end
    if not getconnections then return end
    SwordChanger.hookedParryAll = true
    local conns = getconnections(ReplicatedStorage.Remotes.ParrySuccessAll.OnClientEvent)
    if not conns or #conns == 0 then return end
    local originals = {}
    for _, conn in ipairs(conns) do
        if conn.Function then
            table.insert(originals, conn.Function)
            conn:Disable()
            conn:Disconnect()
        end
    end
    if #originals == 0 then return end
    ReplicatedStorage.Remotes.ParrySuccessAll.OnClientEvent:Connect(function(...)
        local v = { ... }
        if SwordChanger.enabled and SwordChanger.swordName ~= "" and tostring(v[4]) == LocalPlayer.Name and Achaotic_SwordInstances then
            local ok, sw = pcall(function() return Achaotic_SwordInstances:GetSword(SwordChanger.swordName) end)
            if ok and sw and sw.SlashName then
                v[1] = sw.SlashName
                v[3] = sw.Name
            end
        end
        for _, fn in ipairs(originals) do pcall(fn, table.unpack(v)) end
    end)
end

local function EquipSword(swordName)
    if type(swordName) ~= "string" or swordName == "" then return end
    local char = LocalPlayer.Character
    if not char then return end
    pcall(function()
        if Achaotic_SwordInstances and rawget and getupvalues and setupvalue then
            local f = rawget(Achaotic_SwordInstances, "EquipSwordTo")
            if type(f) == "function" then
                local ups = getupvalues(f)
                for i = 1, #ups do
                    if type(ups[i]) == "boolean" then
                        setupvalue(f, i, false)
                        break
                    end
                end

local function EnableSwordChanger(name)

        task.wait(1)

local function MakeDraggable(frame, onDrag)
    local state = { dragging = false, target = nil, dragInput = nil, dragStart = nil, startPos = nil, onDrag = onDrag }
    draggableState[frame] = state
    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            state.dragging = true
            state.dragStart = input.Position
            state.startPos = frame.Position
            state.target = state.startPos
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then state.dragging = false end
            end)
        end
    end)
    frame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            state.dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == state.dragInput and state.dragging then
            local d = input.Position - state.dragStart
            state.target = UDim2.new(
                state.startPos.X.Scale, state.startPos.X.Offset + d.X,
                state.startPos.Y.Scale, state.startPos.Y.Offset + d.Y
            )
        end
    end)
end

RunService.RenderStepped:Connect(function(dt)
    if next(draggableState) == nil then return end
    for frame, state in pairs(draggableState) do
        if not state.target then continue end
        local c = frame.Position
        local a = math.clamp(dt * 25, 0, 1)
        local nx = c.X.Offset + (state.target.X.Offset - c.X.Offset) * a
        local ny = c.Y.Offset + (state.target.Y.Offset - c.Y.Offset) * a
        local settledX = math.abs(nx - state.target.X.Offset) < 0.5
        local settledY = math.abs(ny - state.target.Y.Offset) < 0.5
        if settledX then nx = state.target.X.Offset end
        if settledY then ny = state.target.Y.Offset end
        frame.Position = UDim2.new(state.target.X.Scale, nx, state.target.Y.Scale, ny)
        if state.onDrag then state.onDrag(frame.Position) end
        if settledX and settledY then state.target = nil end
    end
end)

            if eventHash == hash and type(eventKey) == "string" and type(proof) == "string"
                and type(duration) == "number" and typeof(direction) == "CFrame"

            local caller = (not checkcaller) or (function()

        if type(_token) ~= "function" or type(uid) ~= "string" then return nil end

        local meta = getrawmetatable(remote)

        _original[meta] = true
        setreadonly(meta, false)

                return function(_, ...)
                    local a = {...}
                    if _is_valid(a) and not _reverted[self] then _reverted[self] = a end
                    return oldIndex(self, k)(_, table.unpack(a))
                end
            end
            return oldIndex(self, k)
        end
        setreadonly(meta, true)
    end

local function select_target_by_mouse()
    local living = {}
    for _, c in AliveFolder:GetChildren() do
        if c.Name ~= LocalPlayer.Name and c.PrimaryPart then table.insert(living, c) end
    end
    local cam = workspace.CurrentCamera
    local mp = UserInputService:GetMouseLocation()
    local ray = cam:ScreenPointToRay(mp.X, mp.Y)
    local best, bestDot = nil, -math.huge
    for _, c in living do

local function ApplyCurveToCFrame(base)
    local curveType = config.curve_method or "camera"
    if curveType == "camera" then return base end
    if curveType == "dot" then
        local char = LocalPlayer.Character
        local cp = char and (char.PrimaryPart or char:FindFirstChild("HumanoidRootPart"))
        if not cp then return base end
        local nearest, bestD = nil, math.huge
        for _, e in ipairs(AliveFolder:GetChildren()) do
            if e ~= char and e:FindFirstChild("HumanoidRootPart") then
                local d = (cp.Position - e.HumanoidRootPart.Position).Magnitude
                if d < bestD then bestD = d; nearest = e end
            end
        end
        if nearest then
            return CFrame.lookAt(base.Position, nearest.HumanoidRootPart.Position + Vector3.new(0, 1.75, 0))
        end
        return base
    end
    local r = CFrame.new()
    if curveType == "backwards" then r = CFrame.Angles(0, math.rad(180), 0)
    elseif curveType == "down" then r = CFrame.Angles(math.rad(-45), 0, 0)
    elseif curveType == "up" then r = CFrame.Angles(math.rad(45), 0, 0) end
    return base * r
end

local spamCPS = 80
local SPAM_MAX_DT = 0.1
local SPAM_MAX_PER_FRAME = math.ceil(spamCPS * 0.05)
local SPAM_MAX_BURST = math.ceil(spamCPS * 0.1)

local PI = {
    targetRate = spamCPS,
    measuredRate = 0,
    integral = 0,
    integralMin = -2000,
    integralMax = 2000,
    kp = 0.2,
    ki = 0.02,
    window = 0.1,
    windowTime = 0,
    windowFires = 0,
    effectiveRate = spamCPS,
}

local function resetPI()
    PI.targetRate = spamCPS
    PI.effectiveRate = spamCPS
    PI.integral = 0
    PI.measuredRate = 0
    PI.windowTime = 0
    PI.windowFires = 0
    SPAM_MAX_PER_FRAME = math.ceil(spamCPS * 0.05)
    SPAM_MAX_BURST = math.ceil(spamCPS * 0.1)
    SpamAcc = 0
end

local function SendParry()
    return pcall(BallHelper.parry)
end

RunService.PreSimulation:Connect(function(dt)
    if not spamActive then
        SpamAcc = 0
        PI.integral = 0
        PI.windowTime = 0
        PI.windowFires = 0
        return
    end
    if not is_player_alive_and_in_alive() then SpamAcc = 0 return end
    local char = LocalPlayer.Character
    if not char or not char.PrimaryPart then SpamAcc = 0 return end
    if dt > SPAM_MAX_DT then dt = SPAM_MAX_DT end
    if dt < 0 then dt = 0 end

    PI.windowTime = PI.windowTime + dt
    if PI.windowTime >= PI.window then
        local measured = PI.windowFires / PI.windowTime
        PI.measuredRate = PI.measuredRate * 0.6 + measured * 0.4
        local error = PI.targetRate - PI.measuredRate
        PI.integral = math.clamp(
            PI.integral + error * PI.windowTime,
            PI.integralMin,
            PI.integralMax
        )
        local correction = PI.kp * error + PI.ki * PI.integral
        PI.effectiveRate = math.clamp(
            PI.targetRate + correction,
            PI.targetRate * 0.5,
            PI.targetRate * 1.5
        )
        PI.windowTime = 0
        PI.windowFires = 0
    end

    local interval = 1 / PI.effectiveRate
    SpamAcc = SpamAcc + dt
    if SpamAcc > interval * SPAM_MAX_BURST then
        SpamAcc = interval * SPAM_MAX_BURST
    end
    local fires = math.floor(SpamAcc / interval)
    if fires > SPAM_MAX_PER_FRAME then fires = SPAM_MAX_PER_FRAME end
    if fires <= 0 then return end
    SpamAcc = SpamAcc - fires * interval
    if SpamAcc < 0 then SpamAcc = 0 end
    PI.windowFires = PI.windowFires + fires
    for _ = 1, fires do SendParry() end
end)

function BallHelper.ball_curved(ball)
    ball = ball or BallHelper.get_ball()
    if not ball then return false end
    local props = BallHelper.ball.properties
    local zoomies = Ball_Zoomies_Cache[ball] or ball:FindFirstChild("zoomies")
    if not zoomies then return false end
    Ball_Zoomies_Cache[ball] = zoomies
    local velocity = zoomies.VectorVelocity
    local velUnit = velocity.Unit
    local ballPos = (ball == Tracked_Ball and Real_Ball_Position) or ball.Position
    local myPos = LocalPlayer.Character and LocalPlayer.Character.PrimaryPart and LocalPlayer.Character.PrimaryPart.Position
    if not myPos then return false end
    local toMe = (myPos - ballPos).Unit
    local dotProduct = toMe:Dot(velUnit)
    local speed = velocity.Magnitude
    local speedFactor = math.min(speed / 100, 40)
    local accelDir = (velUnit - velocity).Unit
    local crossDot = toMe:Dot(accelDir)
    local dotDifference = dotProduct - crossDot
    local distance = (myPos - ballPos).Magnitude
    local pingMs = Snapshot.ping
    local pingThreshold = 0.5 - (pingMs / 1000)
    local timeToImpact = distance / speed - (pingMs / 1000)
    local minDistance = 15 - math.min(distance / 1000, 15) + speedFactor
    local clampedDot = math.clamp(dotProduct, -1, 1)
    local angleRad = math.rad(math.asin(clampedDot))
    props.lerp_radians = linear_predict(props.lerp_radians, angleRad, 0.8)
    if speed > 100 and timeToImpact > pingMs / 10 then
        minDistance = math.max(minDistance - 15, 15)
    end
    if distance < minDistance then return false end
    if dotDifference < pingThreshold then return true end
    if props.lerp_radians < 0.018 then props.last_warping = tick() end
    if (tick() - props.last_warping) < (timeToImpact / 1.5) then return true end
    if (tick() - props.curving) < (timeToImpact / 1.5) then return true end
    return dotProduct < pingThreshold
end

local function GetAccuracyMultiplier(value)
    return 1.285 * (math.clamp(tonumber(value) or 100, 1, 100) / 100)
end
local accuracyMultiplier = GetAccuracyMultiplier(config.accuracy)

local parryActive = false
local parried_balls = setmetatable({}, { __mode = "k" })
local tracked_balls = setmetatable({}, { __mode = "k" })

local function track_ball(ball)
    if tracked_balls[ball] then return end
    tracked_balls[ball] = true
    ball:GetAttributeChangedSignal("target"):Connect(function()
        parried_balls[ball] = nil
        parryActive = false
    end)
    ball.Destroying:Connect(function()
        parried_balls[ball] = nil
        tracked_balls[ball] = nil
        Ball_Zoomies_Cache[ball] = nil
    end)
end

local function SetAutoParry(enabled)
    config.auto_parry = enabled
    if enabled then
        if Connections.auto_parry then return end
        Connections.auto_parry = RunService.PreSimulation:Connect(function()
            if isInLobby() then return end
            if not is_player_alive_and_in_alive() then return end
            local char = LocalPlayer.Character
            if not char or not char.PrimaryPart then return end
            local props = BallHelper.ball.properties
            local allBalls = {}
            local realBall = nil
            for _, b in ipairs(BallsFolder:GetChildren()) do
                if b:GetAttribute("realBall") then
                    allBalls[#allBalls + 1] = b
                    if not realBall then realBall = b end
                end
            end
            local tb = GetTrainingBallsFolder()
            if tb then
                for _, b in ipairs(tb:GetChildren()) do
                    if b:GetAttribute("realBall") then
                        allBalls[#allBalls + 1] = b
                        if not realBall then realBall = b end
                    end
                end
            end
            local realBallTarget = realBall and realBall:GetAttribute("target")
            local pingMs = Snapshot.ping / 10
            local pingBase = math.clamp(pingMs / 10, 5, 17)
            local accuracyFactor = accuracyMultiplier
            for _, ball in allBalls do
                local zoomies = Ball_Zoomies_Cache[ball] or ball:FindFirstChild("zoomies")
                if not zoomies then continue end
                Ball_Zoomies_Cache[ball] = zoomies
                track_ball(ball)
                if parried_balls[ball] then continue end
                if parryActive then continue end
                local ballTarget = ball:GetAttribute("target")
                local ballPos = (ball == Tracked_Ball and Real_Ball_Position) or ball.Position
                local velocity = zoomies.VectorVelocity
                local distance = (char.PrimaryPart.Position - ballPos).Magnitude
                local rawSpeed = velocity.Magnitude
                local speedAdjusted = math.min(math.max(rawSpeed - 9.5, 0), 650)
                local speedCoefficient = 2.4 + speedAdjusted * 0.002
                local scaledAg = speedCoefficient * accuracyMultiplier
                local targetDistance = pingBase + math.max(rawSpeed / scaledAg, 9.5)
                local isCurving = BallHelper.ball_curved(ball)
                if ball:FindFirstChild("AeroDynamicSlashVFX") then
                    Debris:AddItem(ball.AeroDynamicSlashVFX, 0)
                    props.aerodynamic_time = tick()
                end
                if RuntimeFolder:FindFirstChild("Tornado") then
                    local tTime = RuntimeFolder.Tornado:GetAttribute("TornadoTime") or 1
                    if (tick() - props.aerodynamic_time) < tTime + 0.314159 then continue end
                end
                if realBallTarget == LocalPlayer.Name and isCurving then continue end
                if ballTarget == LocalPlayer.Name and distance <= targetDistance and accuracyFactor * 0.8 then
                    parried_balls[ball] = true
                    parryActive = true
                    BallHelper.parry()
                    local token = tick()
                    task.delay(0.5, function()
                        if tick() - token >= 0.5 then parryActive = false end
                    end)
                end
            end
        end)
    else
        if Connections.auto_parry then
            Connections.auto_parry:Disconnect()
            Connections.auto_parry = nil
        end
        parryActive = false
    end
end

local function ApplyAvatar(character, userId)
    if not character or not userId then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    if character:GetAttribute("_417Applied") == userId then return end
    character:SetAttribute("_417Applied", userId)
    local ok, desc = pcall(function() return Players:GetHumanoidDescriptionFromUserId(userId) end)
    if not ok or not desc then character:SetAttribute("_417Applied", nil) return end
    local ok2, model = pcall(function() return Players:CreateHumanoidModelFromDescription(desc, humanoid.RigType) end)
    if not ok2 or not model then character:SetAttribute("_417Applied", nil) return end
    for _, child in ipairs(character:GetChildren()) do
        if child:IsA("Accessory") or child:IsA("Shirt") or child:IsA("Pants") or child:IsA("ShirtGraphic") or child:IsA("BodyColors") or child:IsA("CharacterMesh") then
            child:Destroy()
        end
    end
    local function getMeshId(part)
        if part:IsA("MeshPart") then return part.MeshId end
        local sm = part:FindFirstChildOfClass("SpecialMesh")
        return sm and sm.MeshId or ""
    end
    for _, child in ipairs(model:GetChildren()) do
        if child:IsA("BasePart") and child.Name ~= "HumanoidRootPart" then
            local mine = character:FindFirstChild(child.Name)
            if mine and getMeshId(child) ~= getMeshId(mine) then
                local sm = mine:FindFirstChildOfClass("SpecialMesh")
                if sm then sm:Destroy() end
                local sm2 = child:FindFirstChildOfClass("SpecialMesh")
                if sm2 then sm2:Clone().Parent = mine end
                if child:IsA("MeshPart") and mine:IsA("MeshPart") then
                    mine.MeshId = child.MeshId
                    mine.TextureID = child.TextureID
                end
            end
        elseif child:IsA("CharacterMesh") then
            child:Clone().Parent = character
        end
    end
    for _, child in ipairs(model:GetChildren()) do
        if child:IsA("Shirt") or child:IsA("Pants") or child:IsA("ShirtGraphic") or child:IsA("BodyColors") then
            child:Clone().Parent = character
        end
    end
    for _, child in ipairs(model:GetChildren()) do
        if child:IsA("Accessory") then
            local clone = child:Clone()
            humanoid:AddAccessory(clone)
            WeldAccessory(clone, character)
        end
    end
    local head = character:FindFirstChild("Head")
    local head2 = model:FindFirstChild("Head")
    if head and head2 then
        local decal = head:FindFirstChildOfClass("Decal")
        local decal2 = head2:FindFirstChildOfClass("Decal")
        if decal2 then
            if decal then decal.Texture = decal2.Texture else decal2:Clone().Parent = head end
        elseif decal then
            decal:Destroy()
        end
    end
    model:Destroy()
    AvatarSpooferState.applied = true
end

    if not userId then warn("[417] Avatar: no userId") return end
    AvatarSpooferState.userId = userId

    pcall(function()
        AvatarSpooferState.connKillFeed = LocalPlayer.PlayerGui.KillFeed.Frame.ChildAdded:Connect(function(child)

            if holder and holder:FindFirstChild("Left") then

end

local function UpdateAbilityBillboard(player, data)
    if player == LocalPlayer then return end
    local char = player.Character
    if not char or not char.Parent or char.Parent ~= workspace.Alive then
        data.gui.Enabled = false
        return
    end
    data.gui.Enabled = AbilityESP.enabled
    if not AbilityESP.enabled then return end

    local ability = player:GetAttribute("EquippedAbility")
    if not ability or ability == "" then
        data.label.Text = string.format('<font color="%s">%s</font>', ORANGE_RGB_STR, player.DisplayName)
        return
    end

    local status, color
    if PassiveSet[ability] then
        status = "Passive"
        color = Color3.fromRGB(150, 150, 255)
    elseif ChargeAbilities[ability] then
        local ci = ChargeAbilities[ability]
        local charges = char:GetAttribute(ci.ChargeAttr) or player:GetAttribute(ci.ChargeAttr) or 0
        status = charges > 0 and "Ready" or "Empty"
        color = charges > 0 and Color3.fromRGB(100, 255, 100) or Color3.fromRGB(255, 100, 100)
    elseif char:GetAttribute("AbilityActive") then
        status = "ACTIVE"
        color = Color3.fromRGB(255, 200, 0)
    else
        local onCd = false
        if TryRequireAbilities() and AbilityESP.abilitiesModule.getAbilityCooldown then
            local ok, cd = pcall(function() return AbilityESP.abilitiesModule.getAbilityCooldown(player, ability) end)
            if ok and cd and cd > 0 then onCd = true end
        end
        if onCd then
            status = "CD"
            color = Color3.fromRGB(255, 180, 100)
        else
            status = "Ready"
            color = Color3.fromRGB(100, 255, 100)
        end
    end

    local statusStr = Color3ToRGBString(color)
    data.label.Text = string.format(
        '<font color="%s">%s</font> [<font color="%s">%s</font> | <font color="%s">%s</font>]',
        ORANGE_RGB_STR, player.DisplayName,
        ORANGE_RGB_STR, ability,
        statusStr, status
    )
end

                            config.avatar_id = tostring(resolved)

    end,
})

pcall(function()
    ReplicatedStorage.Remotes.ParrySuccessAll.OnClientEvent:Connect(function(_, parrier)
        if isInLobby() then return end
        if not is_player_alive_and_in_alive() then return end
        if parrier.Parent and parrier.Parent ~= LocalPlayer.Character then
            if parrier.Parent.Parent ~= workspace.Alive then return end
        end
        local ball = BallHelper.get_ball()
        if not ball or not CurrentTarget or not CurrentTarget.PrimaryPart then return end
        local ballPos = (ball == Tracked_Ball and Real_Ball_Position) or ball.Position
        local myPos = LocalPlayer.Character.PrimaryPart.Position
        local distT = (myPos - CurrentTarget.PrimaryPart.Position).Magnitude
        local distB = (myPos - ballPos).Magnitude
        local toBall = (myPos - ballPos).Unit
        local dot = toBall:Dot(ball.AssemblyLinearVelocity.Unit)
        if distT < 15 and distB < 15 and dot > -0.25 then
            if BallHelper.ball_curved(ball) then BallHelper.parry() end
        end
    end)

    ReplicatedStorage.Remotes.ParrySuccessAll.OnClientEvent:Connect(function(_, parrier)
        if isInLobby() then return end
        local myHrp = LocalPlayer.Character.PrimaryPart
        local ball = BallHelper.get_ball()
        if not ball then return end
        local zoomies = ball:FindFirstChild("zoomies")
        if not zoomies then return end
        local ballPos = (ball == Tracked_Ball and Real_Ball_Position) or ball.Position
        local speed = zoomies.VectorVelocity.Magnitude
        local distance = (LocalPlayer.Character.PrimaryPart.Position - ballPos).Magnitude
        local pingMs = Snapshot.ping
        local speedFactor = math.min(speed / 100, 40)
        local timeToImpact = distance / speed - (pingMs / 1000)
        local isFast = speed > 100
        local minDistance = 15 - math.min(distance / 1000, 15) + speedFactor
        if isFast and timeToImpact > pingMs / 10 then
            minDistance = math.max(minDistance - 15, 15)
        end
        if parrier ~= myHrp and distance > minDistance then
            BallHelper.ball.properties.curving = tick()
        end
    end)
end)
